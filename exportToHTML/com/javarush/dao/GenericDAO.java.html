1    package com.javarush.dao;
2    
3    import  jakarta.persistence.Query;
4    import org.hibernate.SessionFactory;
5    import org.hibernate.Session;
6    import java.util.List;
7    
8    public class GenericDAO<T> {
9    
10       private  final  Class<T> clazz;
11   
12       private SessionFactory sessionFactory;
13   
14       public GenericDAO(Class<T> clazz, SessionFactory sessionFactory) {
15           this.clazz = clazz;
16           this.sessionFactory = sessionFactory;
17       }
18   
19       public T getById(final int id){
20           return (T) getCurrentSession().get(clazz, id);
21       }
22   
23       public  List<T> getItems(int offset, int count){
24           Query query = getCurrentSession().createQuery("from " + clazz.getName(), clazz);
25           query.setFirstResult(offset);
26           query.setMaxResults(count);
27           return  query.getResultList();
28   
29       }
30   
31       public  List<T> findAll(){
32           return getCurrentSession().createQuery("from " + clazz.getName(), clazz).list();
33       }
34   
35   public T save(final T entity){
36       getCurrentSession().saveOrUpdate(entity);
37       return  entity;
38   }
39   
40   
41   public  T update(final T entity){
42       return (T)   getCurrentSession().merge(entity);
43   }
44   
45       public void  delete(final T entity){
46               getCurrentSession().delete(entity);
47       }
48   
49       public void  deleteById(final  int entityId){
50           final T entity = getById(entityId);
51           delete(entity);
52       }
53   
54   
55       protected Session getCurrentSession() {
56           return sessionFactory.getCurrentSession();
57       }
58   
59   }
